#!/usr/bin/env bash
set -euo pipefail
repo=/home/dev/work/EZCorp/extension-v4-independent-audit
source_sha=${1:?full source SHA required}
image=${2:?candidate image required}
receipt=${3:?new receipt directory required}
[[ "$source_sha" =~ ^[0-9a-f]{40}$ ]]
[[ ! -e "$receipt" ]] || { echo "Receipt directory already exists: $receipt" >&2; exit 2; }
cd "$repo"
[[ $(git rev-parse HEAD) == "$source_sha" ]] || { echo "Checked-out source is not requested source" >&2; exit 2; }
mkdir -p "$receipt"
receipt=$(realpath "$receipt")
export PATH=/tmp/ez-extension-bun-1.3.14/bun-linux-x64:/nix/store/vs03s8q30qg698zzpbszk08j4shb0gsl-nodejs-slim-22.22.2/bin:$PATH
unset DOCKER_HOST
export BUILDX_BUILDER=default CONMON=/tmp/ez-audit-ci-conmon
export VERIFY_ARCHIVED_IMAGE_RECEIPT_ROOT="$receipt/build-archive"
source scripts/lib/build-archived-image.sh
ensure_archived_image "$source_sha" "$image" candidate
docker_revision=$(docker image inspect "$image" --format '{{index .Config.Labels "org.opencontainers.image.revision"}}')
[[ "$docker_revision" == "$source_sha" ]] || { echo "Docker image revision did not match requested source" >&2; exit 1; }
docker image inspect "$image" > "$receipt/docker-image-inspect.json"
docker save "$image" | podman --remote=false load > "$receipt/docker-to-native-transfer.log" 2>&1
docker_id=$(docker image inspect "$image" --format '{{.Id}}')
podman_id=$(podman --remote=false image inspect "$image" --format '{{.Id}}')
[[ "${docker_id#sha256:}" == "${podman_id#sha256:}" ]] || { echo "Docker/native Podman image IDs differ" >&2; exit 1; }
podman --remote=false image inspect "$image" > "$receipt/native-podman-image-inspect.json"
printf 'source_commit_sha=%s\nimage=%s\ndocker_image_id=%s\nnative_podman_image_id=%s\n' "$source_sha" "$image" "$docker_id" "$podman_id" > "$receipt/provenance.txt"
