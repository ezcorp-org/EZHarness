#!/usr/bin/env bash
set -euo pipefail
repo=/home/dev/work/EZCorp/extension-v4-independent-audit
work="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
receipt="${EZ_R4_RECEIPT_DIR:?}"
expected=9ca275838faf30666da5dba1c0eba141dd053050
image=localhost/ezcorp-extension-v4:shipping-9ca27583
export PATH=/tmp/ez-extension-bun-1.3.14/bun-linux-x64:/nix/store/vs03s8q30qg698zzpbszk08j4shb0gsl-nodejs-slim-22.22.2/bin:$PATH
export DOCKER_HOST=unix:///run/user/1001/podman/podman.sock
cd "$repo"
actual="$(git rev-parse HEAD)"
[[ "$actual" == "$expected" ]]
git diff --quiet -- scripts/verify-production-image-lifecycle.sh scripts/verify-shipping-runtime-resources.ts scripts/lib/shipping-runtime-cycle-conversation.ts scripts/lib/production-lifecycle-client.ts scripts/lib/shipping-runtime-resource-config.ts web/src/lib/server/security/payload.ts
image_metadata="$(docker image inspect "$image" --format 'image_id={{.Id}} revision={{index .Config.Labels "org.opencontainers.image.revision"}}')"
printf '%s\n' "$image_metadata" | grep --line-buffered -F "revision=$expected" > /dev/null
printf 'controller_start=%s\nsource_head=%s\n%s\n' "$(date --iso-8601=seconds)" "$actual" "$image_metadata" > "$receipt/launch-proof.txt"
sha256sum scripts/verify-production-image-lifecycle.sh scripts/verify-shipping-runtime-resources.ts scripts/lib/shipping-runtime-cycle-conversation.ts scripts/lib/production-lifecycle-client.ts scripts/lib/shipping-runtime-resource-config.ts web/src/lib/server/security/payload.ts "$work/run-r4.sh" "$work/execute.sh" > "$receipt/source-hashes-at-launch.sha256"
state="$(mktemp -d /tmp/r4-XXXXXX)"
printf 'state_root=%s\n' "$state" > "$receipt/state-path.txt"
bun --version > "$receipt/tool-versions.txt"
node --version >> "$receipt/tool-versions.txt"
set +e
timeout --foreground --kill-after=30s 2500s env EZ_PRODUCTION_IMAGE="$image" EZ_PRODUCTION_RECEIPT_DIR="$receipt/runtime" EZ_PRODUCTION_STATE_DIR="$state" EZ_PRODUCTION_PORT=4294 EZ_PRODUCTION_COMPOSE_PROJECT=ez-r4-canonical-9ca27583 EZ_PRODUCTION_APP_CONTAINER=ez-r4-canonical-9ca27583-app EZ_PRODUCTION_APP_UID=0 EZ_PRODUCTION_APP_GID=0 bash "$repo/scripts/verify-production-image-lifecycle.sh" -- "$work/run-r4.sh" > "$receipt/launcher.log" 2>&1
run_result=$?
set -e
printf 'canonical_r4_exit=%s\n' "$run_result" > "$receipt/exit.txt"
exit "$run_result"
