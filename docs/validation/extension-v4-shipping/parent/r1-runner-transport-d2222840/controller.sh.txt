#!/usr/bin/env bash
set -euo pipefail
cd /home/dev/work/EZCorp/extension-v4-independent-audit
receipt=${1:?Existing parent receipt required}
export PATH=/tmp/ez-extension-bun-1.3.14/bun-linux-x64:/nix/store/vs03s8q30qg698zzpbszk08j4shb0gsl-nodejs-slim-22.22.2/bin:$PATH
source_file=scripts/lib/production-lifecycle-client.ts
restore() { cp "$receipt/final-inputs/$source_file.txt" "$source_file"; }
finish() { rc=$?; trap - EXIT; restore; printf '%s\n' "$rc" > "$receipt/controller.exit"; exit "$rc"; }
trap finish EXIT
cp "$0" "$receipt/controller.sh.txt"
python3 - "$source_file" "$receipt" <<'PY'
from pathlib import Path
import sys
p=Path(sys.argv[1]); s=p.read_text()
replacements={
'readFile(required("EZ_EXTENSION_RUNNER_TOKEN_FILE"), "utf8")':'readFile(join(required("EZ_PRODUCTION_RUN_ROOT"), "token"), "utf8")',
'unix: required("EZ_EXTENSION_RUNNER_SOCKET")':'unix: join(required("EZ_PRODUCTION_RUN_ROOT"), "socket", "runner.sock")',
}
for before,after in replacements.items():
 assert s.count(before)==1,before
 s=s.replace(before,after)
p.write_text(s)
Path(sys.argv[2],'old-path-control.ts.txt').write_text(s)
PY
set +e
bun test --timeout 30000 ./src/__tests__/production-image-lifecycle-launch.integration.test.ts > "$receipt/old-path-red.log" 2>&1
red_rc=$?
set -e
printf '%s\n' "$red_rc" > "$receipt/old-path-red.exit"
restore
sha256sum -c "$receipt/final-inputs.sha256" > "$receipt/restored-input-check.txt"
[[ "$red_rc" -eq 1 ]]
rg -q 'ENOENT: no such file or directory' "$receipt/old-path-red.log"
rg -q 'ez-r4-runner-private-socket-regression-.*/token' "$receipt/old-path-red.log"
set +e
bun test --timeout 30000 ./src/__tests__/production-image-lifecycle-launch.integration.test.ts > "$receipt/green.log" 2>&1
green_rc=$?
set -e
printf '%s\n' "$green_rc" > "$receipt/green.exit"
[[ "$green_rc" -eq 0 ]]
set +e
bun run typecheck > "$receipt/typecheck.log" 2>&1
types_rc=$?
set -e
printf '%s\n' "$types_rc" > "$receipt/typecheck.exit"
sha256sum -c "$receipt/final-inputs.sha256" > "$receipt/final-input-check.txt"
[[ "$types_rc" -eq 0 ]]
