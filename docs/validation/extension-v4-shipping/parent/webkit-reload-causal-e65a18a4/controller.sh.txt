#!/usr/bin/env bash
set -uo pipefail
repo=/home/dev/work/EZCorp/extension-v4-independent-audit
lock="$repo/.cache/validation-heavy.lock"
if [[ ${1:-} != --locked ]]; then
  exec flock --close "$lock" "$0" --locked
fi
export PATH=/tmp/ez-extension-bun-1.3.14/bun-linux-x64:/nix/store/vs03s8q30qg698zzpbszk08j4shb0gsl-nodejs-slim-22.22.2/bin:$PATH
export CONMON=/tmp/ez-audit-ci-conmon
export PW_TEST_CONNECT_WS_ENDPOINT=ws://127.0.0.1:3333/
export PW_TEST_CONNECT_EXPOSE_NETWORK='<loopback>'
cd "$repo"
receipt="$repo/.cache/terra-shipping/parent/webkit-reload-causal-$(date -u +%Y%m%dT%H%M%SZ)"
mkdir -p "$receipt"
source_file=web/e2e/real-auth/extension-lifecycle-flow.spec.ts
expected_sha=e65a18a442d021364c62a2bb8b3c5d5d3eb8a090454436ec7fd7ef21d01d8e8e
actual_sha=$(sha256sum "$source_file" | awk '{print $1}')
if [[ "$actual_sha" != "$expected_sha" ]]; then
  printf 'expected-source-sha=%s\nactual-source-sha=%s\n' "$expected_sha" "$actual_sha" > "$receipt/source-mismatch.txt"
  exit 3
fi
saved="$receipt/source-before-red.spec.ts"
cp "$source_file" "$saved"
restore() { cp "$saved" "$source_file"; }
trap restore EXIT
trap 'restore; exit 130' INT HUP TERM
printf 'source-head=%s\nsource-sha=%s\ncontroller-pid=%s\n' "$(git rev-parse HEAD)" "$actual_sha" "$$" > "$receipt/provenance.txt"
python3 - "$source_file" <<'PY'
import sys
path = sys.argv[1]
text = open(path).read()
needle = '''      if (pendingBuildInspection?.request === request) pendingBuildInspection.cancellationObserved = true;
      return;
    }
    activeRuntimeEventRequests.delete(request);'''
replacement = '''      if (pendingBuildInspection?.request === request) pendingBuildInspection.cancellationObserved = true;
      // Controlled red: exact captured cancellation remains strict.
    }
    activeRuntimeEventRequests.delete(request);'''
if text.count(needle) != 1:
    raise SystemExit('controlled-red patch anchor is not unique')
open(path, 'w').write(text.replace(needle, replacement))
PY
sha256sum "$source_file" > "$receipt/red-source.sha256"
set +e
(
  cd web
  CI=1 PI_E2E_REAL=1 PI_E2E_REAL_BROWSER_PROJECTS=webkit EZCORP_E2E_EVIDENCE=1 \
    bunx playwright test --config playwright.real.config.ts --project=webkit \
    e2e/real-auth/extension-lifecycle-flow.spec.ts --grep "reloads an observed pending browser build" \
    --output="../${receipt#$repo/}/red-results"
) > "$receipt/red-webkit.log" 2>&1
red=$?
set -e
printf '%s\n' "$red" > "$receipt/red.exit"
if [[ "$red" -eq 0 ]] || ! grep -Fq 'failedApiRequests' "$receipt/red-webkit.log" || ! grep -Fq 'Load request cancelled' "$receipt/red-webkit.log"; then
  exit 2
fi
restore
restored_sha=$(sha256sum "$source_file" | awk '{print $1}')
printf '%s\n' "$restored_sha" > "$receipt/restored-source.sha256"
if [[ "$restored_sha" != "$expected_sha" ]]; then exit 4; fi
set +e
(
  cd web
  CI=1 PI_E2E_REAL=1 PI_E2E_REAL_BROWSER_PROJECTS=webkit EZCORP_E2E_EVIDENCE=1 \
    bunx playwright test --config playwright.real.config.ts --project=webkit \
    e2e/real-auth/extension-lifecycle-flow.spec.ts --grep "reloads an observed pending browser build" \
    --output="../${receipt#$repo/}/green-results"
) > "$receipt/green-webkit.log" 2>&1
green=$?
set -e
printf '%s\n' "$green" > "$receipt/green.exit"
all_engines=0
for engine in chromium firefox webkit; do
  if [[ "$engine" == webkit ]]; then
    extra=(PW_TEST_CONNECT_WS_ENDPOINT=ws://127.0.0.1:3333/ PW_TEST_CONNECT_EXPOSE_NETWORK='<loopback>')
  else
    extra=(-u PW_TEST_CONNECT_WS_ENDPOINT -u PW_TEST_CONNECT_EXPOSE_NETWORK)
  fi
  set +e
  (
    cd web
    env "${extra[@]}" CI=1 PI_E2E_REAL=1 PI_E2E_REAL_BROWSER_PROJECTS="$engine" EZCORP_E2E_EVIDENCE=1 \
      bunx playwright test --config playwright.real.config.ts --project="$engine" \
      e2e/real-auth/extension-lifecycle-flow.spec.ts \
      --output="../${receipt#$repo/}/full-${engine}-results"
  ) > "$receipt/full-${engine}.log" 2>&1
  engine_exit=$?
  set -e
  printf '%s\t%s\n' "$engine" "$engine_exit" >> "$receipt/full-engines.tsv"
  [[ "$engine_exit" -eq 0 ]] || all_engines=1
done
set +e
bash scripts/typecheck.sh > "$receipt/canonical-e2e-typecheck.log" 2>&1
typecheck=$?
set -e
printf '%s\n' "$typecheck" > "$receipt/canonical-e2e-typecheck.exit"
[[ "$green" -eq 0 && "$all_engines" -eq 0 && "$typecheck" -eq 0 ]]
