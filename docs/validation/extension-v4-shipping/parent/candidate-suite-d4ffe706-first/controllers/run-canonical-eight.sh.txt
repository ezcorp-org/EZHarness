#!/usr/bin/env bash
set -euo pipefail
repo=/home/dev/work/EZCorp/extension-v4-independent-audit
source_sha=${1:?full source SHA required}
image=${2:?candidate image required}
receipt=${3:?new receipt directory required}
image_source_sha=${4:-$source_sha}
[[ "$source_sha" =~ ^[0-9a-f]{40}$ ]]
[[ "$image_source_sha" =~ ^[0-9a-f]{40}$ ]]
[[ ! -e "$receipt" ]] || { echo "Receipt directory already exists: $receipt" >&2; exit 2; }
cd "$repo"
mkdir -p "$receipt"
receipt=$(realpath "$receipt")
source "$repo/.cache/terra-shipping/parent/new-payload-final/verify-checkout-image-source.sh" || exit $?
assert_checkout_and_image_source "$source_sha" "$image_source_sha" "$receipt" || exit $?
assert_tracked_product_inputs_clean "$source_sha" "$receipt" || exit $?

export PATH=/tmp/ez-extension-bun-1.3.14/bun-linux-x64:/nix/store/vs03s8q30qg698zzpbszk08j4shb0gsl-nodejs-slim-22.22.2/bin:$PATH
export CONMON=/tmp/ez-audit-ci-conmon DOCKER_HOST=unix:///run/user/1001/podman/podman.sock BUILDX_BUILDER=default
native_revision=$(podman --remote=false image inspect "$image" --format '{{index .Config.Labels "org.opencontainers.image.revision"}}')
[[ "$native_revision" == "$image_source_sha" ]] || { echo "Native candidate revision did not match requested source" >&2; exit 1; }
podman --remote=false image inspect "$image" > "$receipt/native-podman-image-inspect.json"
export VERIFY_UPGRADE_CANDIDATE_SOURCE="$image_source_sha" EZ_SHIPPING_CANDIDATE_IMAGE="$image" EZ_SHIPPING_RECEIPT_ROOT="$receipt/runtime"
export EZ_SHIPPING_APP_UID=0 EZ_SHIPPING_APP_GID=0 EZ_SHIPPING_RUNNER_APP_UID=1001
export VERIFY_UPGRADE_PREVIOUS_IMAGE=localhost/ezcorp-extension-v4:parent-baseline-3ec53eaa
export VERIFY_LEGACY_ADOPTION_BASE_IMAGE=localhost/ezcorp-extension-v4:legacy-main-537f074e
export VERIFY_LEGACY_ADOPTION_IMAGE=localhost/ezcorp-extension-v4:legacy-main-537f074e-derived
printf 'checkout_source_sha=%s\nimage_source_sha=%s\nimage=%s\nnative_revision=%s\n' "$source_sha" "$image_source_sha" "$image" "$native_revision" > "$receipt/provenance.txt"
if bash scripts/verify-shipping-production-suite.sh; then
  rc=0
else
  rc=$?
fi
printf '%s\n' "$rc" > "$receipt/exit"
exit "$rc"
