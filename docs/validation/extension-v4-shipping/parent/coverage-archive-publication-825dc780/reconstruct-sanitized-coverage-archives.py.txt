#!/usr/bin/env python3
"""Reconstruct two coverage tar.zst archives with one fixture ID redacted per tar.

This is a publication transformation. It preserves logical tar members and metadata,
not the old compressed-container identity. It never writes the original fixture ID.
"""
from __future__ import annotations
import copy
from hashlib import sha256
import io
import json
from pathlib import Path
import re
import subprocess
import sys
import tarfile

TARGET = "final-coverage-curated-19d/host/result_1423"
FIELD = re.compile(rb'("tokenActorExtensionId"\s*:\s*")[^"]+("\s*[,}])')
REPLACEMENT = rb'\1<fixture-extension-id>\2'
CHUNK = 1024 * 1024


def die(message: str) -> None:
    raise RuntimeError(message)


def header(info: tarfile.TarInfo) -> dict[str, object]:
    return {
        "name": info.name,
        "type": info.type.hex(),
        "linkname": info.linkname,
        "mode": info.mode,
        "uid": info.uid,
        "gid": info.gid,
        "size": info.size,
        "mtime": info.mtime,
        "uname": info.uname,
        "gname": info.gname,
        "devmajor": info.devmajor,
        "devminor": info.devminor,
        "paxHeaders": info.pax_headers,
    }


def digest_reader(reader) -> str:
    digest = sha256()
    while True:
        chunk = reader.read(CHUNK)
        if not chunk:
            return digest.hexdigest()
        digest.update(chunk)


def inventory(path: Path) -> list[dict[str, object]]:
    process = subprocess.Popen(["zstd", "--decompress", "--stdout", "--quiet", "--", str(path)], stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)
    records: list[dict[str, object]] = []
    try:
        with tarfile.open(fileobj=process.stdout, mode="r|") as archive:
            for info in archive:
                record = header(info)
                if info.isreg():
                    reader = archive.extractfile(info)
                    if reader is None:
                        die(f"unreadable regular member: {info.name}")
                    record["payloadSha256"] = digest_reader(reader)
                records.append(record)
    finally:
        if process.stdout is not None:
            process.stdout.close()
        if process.wait() != 0:
            die(f"cannot decompress {path.name}")
    return records


def sanitized_target(body: bytes) -> bytes:
    lines = body.splitlines(keepends=True)
    if len(lines) < 12:
        die("target result has fewer than 12 lines")
    updated, replacements = FIELD.subn(REPLACEMENT, lines[11], count=1)
    if replacements != 1:
        die("target line does not contain exactly one tokenActorExtensionId field")
    lines[11] = updated
    result = b"".join(lines)
    if result == body:
        die("target transformation did not change bytes")
    return result


def reconstruct(source: Path, target: Path) -> dict[str, object]:
    source_digest = sha256(source.read_bytes()).hexdigest()
    source_bytes = source.stat().st_size
    target.parent.mkdir(parents=True, exist_ok=True)
    decompressor = subprocess.Popen(["zstd", "--decompress", "--stdout", "--quiet", "--", str(source)], stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)
    with target.open("xb") as target_file:
        compressor = subprocess.Popen(["zstd", "--compress", "--ultra", "-19", "--threads=1", "--stdout", "--quiet"], stdin=subprocess.PIPE, stdout=target_file, stderr=subprocess.DEVNULL)
        transformed = 0
        try:
            with tarfile.open(fileobj=decompressor.stdout, mode="r|") as incoming, tarfile.open(fileobj=compressor.stdin, mode="w|") as outgoing:
                for info in incoming:
                    copied = copy.copy(info)
                    copied.pax_headers = dict(info.pax_headers)
                    if info.isreg():
                        reader = incoming.extractfile(info)
                        if reader is None:
                            die(f"unreadable regular member: {info.name}")
                        if info.name == TARGET:
                            body = reader.read()
                            body = sanitized_target(body)
                            copied.size = len(body)
                            outgoing.addfile(copied, io.BytesIO(body))
                            transformed += 1
                        else:
                            outgoing.addfile(copied, reader)
                    else:
                        outgoing.addfile(copied)
        finally:
            if compressor.stdin is not None and not compressor.stdin.closed:
                compressor.stdin.close()
            if decompressor.stdout is not None:
                decompressor.stdout.close()
            compress_rc = compressor.wait()
            decompress_rc = decompressor.wait()
    if decompress_rc != 0 or compress_rc != 0:
        die(f"reconstruction stream failed: decompress={decompress_rc} compress={compress_rc}")
    if transformed != 1:
        die(f"expected one transformed member, got {transformed}")
    return {
        "sourceArchiveSha256": source_digest,
        "sourceArchiveBytes": source_bytes,
        "publicArchiveSha256": sha256(target.read_bytes()).hexdigest(),
        "publicArchiveBytes": target.stat().st_size,
        "transformedMembers": transformed,
    }


def expected_target(source: Path) -> tuple[int, str]:
    process = subprocess.Popen(["zstd", "--decompress", "--stdout", "--quiet", "--", str(source)], stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)
    expected: tuple[int, str] | None = None
    try:
        with tarfile.open(fileobj=process.stdout, mode="r|") as archive:
            for info in archive:
                if not info.isreg():
                    continue
                reader = archive.extractfile(info)
                if reader is None:
                    die(f"unreadable regular member: {info.name}")
                if info.name == TARGET:
                    body = sanitized_target(reader.read())
                    if expected is not None:
                        die("target result appears more than once")
                    expected = len(body), sha256(body).hexdigest()
                else:
                    digest_reader(reader)
    finally:
        if process.stdout is not None:
            process.stdout.close()
        if process.wait() != 0:
            die(f"cannot decompress {source.name}")
    if expected is None:
        die("target result was absent")
    return expected


def compare(source: Path, target: Path) -> dict[str, int]:
    expected_size, expected_sha256 = expected_target(source)
    before, after = inventory(source), inventory(target)
    if len(before) != len(after):
        die("member count changed")
    unchanged_regular = 0
    transformed = 0
    for old, new in zip(before, after, strict=True):
        if old["name"] != new["name"] or old["type"] != new["type"]:
            die("member ordering or type changed")
        if old["name"] == TARGET and "payloadSha256" in old:
            old_header = dict(old); new_header = dict(new)
            old_header.pop("size"); new_header.pop("size")
            old_header.pop("payloadSha256"); new_header.pop("payloadSha256")
            if old_header != new_header:
                die("target metadata changed beyond size")
            if new["size"] != expected_size or new.get("payloadSha256") != expected_sha256:
                die("target payload is not the required sanitized replacement")
            transformed += 1
            continue
        if old != new:
            die(f"unchanged member changed: {old['name']}")
        if "payloadSha256" in old:
            unchanged_regular += 1
    if transformed != 1:
        die(f"expected one transformed target in comparison, got {transformed}")
    return {
        "memberCount": len(before),
        "unchangedRegularPayloadCount": unchanged_regular,
        "transformedMemberCount": transformed,
        "sanitizedTargetBytes": expected_size,
        "sanitizedTargetSha256": expected_sha256,
    }


def main() -> None:
    private_root, public_root, receipt = map(Path, sys.argv[1:4])
    names = ["final-coverage-curated-3a56a869.tar.zst", "final-coverage-curated-ea445e9e-plus4d.tar.zst"]
    archives = []
    for name in names:
        source = private_root / name
        target = public_root / name
        if not source.is_file() or target.exists():
            die(f"invalid archive paths for {name}")
        identity = reconstruct(source, target)
        identity.update(compare(source, target))
        identity["archive"] = name
        archives.append(identity)
    receipt.parent.mkdir(parents=True, exist_ok=True)
    receipt.write_text(json.dumps({
        "scope": "sanitized-publication-reconstruction",
        "targetMember": TARGET,
        "replacement": "<fixture-extension-id>",
        "archives": archives,
    }, indent=2) + "\n")

if __name__ == "__main__":
    main()
