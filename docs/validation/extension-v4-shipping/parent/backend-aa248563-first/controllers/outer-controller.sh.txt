#!/usr/bin/env bash
# Caller owns the one shared heavy-validation lock.
set -euo pipefail
repo=/home/dev/work/EZCorp/extension-v4-independent-audit
cd "$repo"
expected=${1:?Exact integrated source commit}
receipt=${2:?New absolute receipt directory}
[[ "$receipt" == /* && ! -e "$receipt" ]] || exit 2
[[ "$(git rev-parse HEAD)" == "$expected" ]] || exit 3
git diff --quiet HEAD -- src web scripts packages .github Dockerfile .dockerignore package.json bun.lock manifest.lock.json docs/extensions docs/development-lifecycle.md
[[ -z "$(git ls-files --others --exclude-standard -- src web scripts packages .github docs/extensions)" ]] || exit 4
mkdir -p "$receipt/inputs"
export PATH=/tmp/ez-extension-bun-1.3.14/bun-linux-x64:/nix/store/vs03s8q30qg698zzpbszk08j4shb0gsl-nodejs-slim-22.22.2/bin:$PATH
export CONMON=/tmp/ez-audit-ci-conmon
export BUN_RUNTIME_TRANSPILER_CACHE_PATH=0
cp "${BASH_SOURCE[0]}" "$receipt/controller.sh.txt"
printf '%s\n' "$expected" > "$receipt/source-commit.txt"
date -u +%Y-%m-%dT%H:%M:%SZ > "$receipt/started-at.txt"
finish() {
  local rc=$?
  trap - EXIT
  printf '%s\n' "$rc" > "$receipt/controller.exit"
  date -u +%Y-%m-%dT%H:%M:%SZ > "$receipt/completed-at.txt"
  exit "$rc"
}
trap finish EXIT
for input in src/db/data-path.ts src/db/connection.ts src/memory/embeddings.ts src/__tests__/db-data-path.test.ts src/__tests__/memory-embeddings-state.test.ts scripts/coverage-thresholds.json scripts/typecheck.sh scripts/test-coverage.sh scripts/test.sh scripts/verify-shipping-embeddings.ts .cache/terra-shipping/backend/run-full-backend.sh; do
  mkdir -p "$receipt/inputs/$(dirname "$input")"
  cp "$input" "$receipt/inputs/$input.txt"
  sha256sum "$input" >> "$receipt/input-hashes.txt"
done
: > "$receipt/results.tsv"
failed=0
run_check() {
  local name=$1
  shift
  local start rc
  start=$(date +%s)
  if "$@" > "$receipt/$name.log" 2>&1; then rc=0; else rc=$?; fi
  printf '%s\t%s\t%s\n' "$name" "$rc" "$(( $(date +%s) - start ))" >> "$receipt/results.tsv"
  cat "$receipt/results.tsv"
  if [[ "$rc" -ne 0 ]]; then failed=1; fi
}
run_check backend-complete env EZ_BACKEND_LOCK_HELD=1 BASE_REF=origin/main bash .cache/terra-shipping/backend/run-full-backend.sh "$receipt/backend"
run_check authored-whitespace git diff --check 8a47c37e -- src web scripts packages
[[ "$(git rev-parse HEAD)" == "$expected" ]] || exit 5
git diff --quiet HEAD -- src web scripts packages .github Dockerfile .dockerignore package.json bun.lock manifest.lock.json docs/extensions docs/development-lifecycle.md
exit "$failed"
