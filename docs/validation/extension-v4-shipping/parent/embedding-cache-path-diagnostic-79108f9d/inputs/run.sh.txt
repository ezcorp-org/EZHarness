#!/usr/bin/env bash
set -euo pipefail
ROOT=/home/dev/work/EZCorp/extension-v4-independent-audit
LOCK="$ROOT/.cache/validation-heavy.lock"
R="${1:?receipt}"
IMAGE=ezcorp:embedding-cache-final-79108f9deb12
state="$R/state"
cleanup=0
finish() {
  local rc=$?
  set +e
  rm -rf "$state" || cleanup=1
  printf 'command_exit=%s\ncleanup_exit=%s\nfinal_exit=%s\n' "$rc" "$cleanup" "$([[ "$rc" -eq 0 && "$cleanup" -eq 0 ]] && echo 0 || echo 1)" > "$R/exit.txt"
  exit "$([[ "$rc" -eq 0 && "$cleanup" -eq 0 ]] && echo 0 || echo 1)"
}
trap finish EXIT
unset DOCKER_HOST DOCKER_CONTEXT CONTAINER_HOST PODMAN_HOST CONTAINER_CONNECTIONS_CONF
export PATH="/tmp/ez-extension-bun-1.3.14/bun-linux-x64:/nix/store/vs03s8q30qg698zzpbszk08j4shb0gsl-nodejs-slim-22.22.2/bin:$PATH"
docker image inspect "$IMAGE" --format 'id={{.Id}} revision={{index .Config.Labels "org.opencontainers.image.revision"}} user={{.Config.User}}' > "$R/image-identity.txt"
set +e
timeout 120s docker run --rm --user 1001:1001 --workdir /app \
  -v "$state/app-data:/app/data" \
  -v "$R/inputs/diagnostic.mjs.txt:/diagnostic.mjs:ro" \
  --entrypoint bun "$IMAGE" /diagnostic.mjs > "$R/diagnostic.json" 2> "$R/diagnostic.stderr.log"
rc=$?
set -e
exit "$rc"
