#!/usr/bin/env python3
from __future__ import annotations
import gzip, hashlib, json, stat, sys, tarfile, zipfile
from collections import deque
from pathlib import Path, PurePosixPath
MAX_MEMBERS=20_000
MAX_BYTES=512*1024*1024
def fail(message): raise ValueError(message)
def safe_parts(name):
 name=name.replace('\\','/')
 path=PurePosixPath(name)
 if path.is_absolute() or not path.parts or any(x in ('','.','..') or ':' in x for x in path.parts): fail(f'unsafe archive member path: {name!r}')
 return path.parts
def kind(name):
 name=name.lower()
 if name.endswith(('.tar.gz','.tgz')): return 'tar.gz'
 if name.endswith('.tar'): return 'tar'
 if name.endswith('.zip'): return 'zip'
 if name.endswith('.gz'): return 'gz'
 return None
def write_member(root,serial,entry,parts,payload):
 target=root/f'archive-{serial:06d}'/f'member-{entry:06d}'/Path(*parts)
 target.parent.mkdir(parents=True,exist_ok=True)
 if root.resolve() not in target.resolve().parents: fail('archive extraction escaped scan root')
 target.write_bytes(payload); return target
def add(queue,records,root,source,chain,member,serial,payload,totals):
 totals['members']+=1; totals['bytes']+=len(payload); totals['entries']+=1
 if totals['members']>MAX_MEMBERS or totals['bytes']>MAX_BYTES: fail('archive expansion limit exceeded')
 parts=safe_parts(member); target=write_member(root,serial,totals['entries'],parts,payload)
 record={'source':source,'archiveChain':chain,'member':'/'.join(parts),'expandedPath':str(target.relative_to(root)),'payloadSha256':hashlib.sha256(payload).hexdigest()}; records.append(record)
 if kind(record['member']): queue.append((target,source,chain+[record['member']]))
def expand(path,source,chain,root,queue,records,serial,totals):
 parent_member=chain[-1] if chain else ""
 archive_kind=kind(parent_member or path.name)
 if archive_kind=='gz': add(queue,records,root,source,chain,(parent_member or path.name)[:-3],serial,gzip.decompress(path.read_bytes()),totals); return
 if archive_kind in ('tar','tar.gz'):
  with tarfile.open(path,'r:gz' if archive_kind=='tar.gz' else 'r:') as archive:
   for info in archive.getmembers():
    if info.isdir(): continue
    if not info.isreg(): fail(f'non-regular tar member: {info.name!r}')
    member='/'.join(safe_parts(info.name)); reader=archive.extractfile(info)
    if reader is None: fail(f'unreadable tar member: {member!r}')
    add(queue,records,root,source,chain,member,serial,reader.read(),totals)
  return
 if archive_kind=='zip':
  with zipfile.ZipFile(path) as archive:
   for info in archive.infolist():
    if info.is_dir(): continue
    if stat.S_ISLNK(info.external_attr>>16): fail(f'symlink zip member: {info.filename!r}')
    add(queue,records,root,source,chain,'/'.join(safe_parts(info.filename)),serial,archive.read(info),totals)
  return
 fail(f'unsupported archive: {path}')
def main():
 staged_root,receipt=map(Path,sys.argv[1:3]); validation=staged_root/'docs'/'validation'; output=staged_root/'expanded-validation-review'; output.mkdir(parents=True,exist_ok=True)
 queue=deque()
 for evidence in sorted(x for x in validation.glob('extension-v4*') if x.is_dir()):
  for source in evidence.rglob('*'):
   if source.is_file() and kind(source.name): queue.append((source,str(source.relative_to(staged_root)),[]))
 records=[]; totals={'members':0,'bytes':0,'entries':0}; serial=0
 while queue:
  path,source,chain=queue.popleft(); serial+=1; expand(path,source,chain,output,queue,records,serial,totals)
 (receipt/'expanded-inputs.json').write_text(json.dumps({'archives':records,'memberCount':totals['members'],'bytes':totals['bytes']},indent=2)+'\n')
if __name__=='__main__': main()
