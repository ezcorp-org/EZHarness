#!/usr/bin/env bash
set -euo pipefail
wt=/home/dev/work/EZCorp/extension-v4-badge-contrast
receipt=${1:?receipt}
cd "$wt"
export PATH=/tmp/ez-extension-bun-1.3.14/bun-linux-x64:/nix/store/vs03s8q30qg698zzpbszk08j4shb0gsl-nodejs-slim-22.22.2/bin:$PATH
printf 'source=%s\n' "$(git rev-parse HEAD)" > "$receipt/provenance.txt"
printf 'tree=%s\n' "$(git rev-parse HEAD^{tree})" >> "$receipt/provenance.txt"
printf 'bun=%s\nnode=%s\n' "$(bun --version)" "$(node --version)" >> "$receipt/provenance.txt"
date -u +%Y-%m-%dT%H:%M:%SZ > "$receipt/started-at.txt"
git diff -- web/e2e/extension-deep-links.spec.ts 'web/src/routes/(app)/extensions/[id]/+page.svelte' > "$receipt/patch-before.diff"
cp 'web/src/routes/(app)/extensions/[id]/+page.svelte' "$receipt/page-fixed.svelte.txt"
python3 - <<'PY'
from pathlib import Path
p=Path('web/src/routes/(app)/extensions/[id]/+page.svelte')
s=p.read_text()
s=s.replace('bg-green-300/40 px-2 py-0.5 text-xs text-green-900 dark:bg-green-900/40 dark:text-green-400', 'bg-green-900/40 px-2 py-0.5 text-xs text-green-400')
s=s.replace('bg-amber-300/40 px-2 py-0.5 text-xs text-amber-900 dark:bg-amber-900/40 dark:text-amber-400', 'bg-amber-900/40 px-2 py-0.5 text-xs text-amber-400')
s=s.replace("bg-red-300/40 text-red-900 dark:bg-red-900/40 dark:text-red-400", "bg-red-900/40 text-red-400")
p.write_text(s)
PY
set +e
( cd web && EZCORP_E2E_EVIDENCE=1 bunx playwright test e2e/extension-deep-links.spec.ts --grep 'header status badges' ) > "$receipt/red.log" 2>&1
red=$?
set -e
printf '%s\n' "$red" > "$receipt/red.exit"
cp "$receipt/page-fixed.svelte.txt" 'web/src/routes/(app)/extensions/[id]/+page.svelte'
set +e
( cd web && EZCORP_E2E_EVIDENCE=1 bunx playwright test e2e/extension-deep-links.spec.ts ) > "$receipt/green.log" 2>&1
green=$?
set -e
printf '%s\n' "$green" > "$receipt/green.exit"
git diff --check > "$receipt/diff-check.log" 2>&1
printf '%s\n' "$?" > "$receipt/diff-check.exit"
date -u +%Y-%m-%dT%H:%M:%SZ > "$receipt/completed-at.txt"
if [[ "$red" -eq 0 || "$green" -ne 0 ]]; then exit 1; fi
