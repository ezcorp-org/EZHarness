#!/usr/bin/env bash
set -euo pipefail
repo=/home/dev/work/EZCorp/extension-v4-independent-audit
receipt=${1:?receipt required}
cd "$repo"
unset DOCKER_HOST DOCKER_CONTEXT CONTAINER_HOST CONTAINER_CONNECTION CONTAINER_CONNECTIONS_CONF PODMAN_HOST
export PATH=/tmp/ez-audit-ci-conmon:/tmp/ez-extension-bun-1.3.14/bun-linux-x64:/nix/store/vs03s8q30qg698zzpbszk08j4shb0gsl-nodejs-slim-22.22.2/bin:$PATH
export CONMON=/tmp/ez-audit-ci-conmon BUN_RUNTIME_TRANSPILER_CACHE_PATH=0
printf 'head=%s\nimage=%s\napp_uid=1001\napp_gid=%s\nrunner_uid=1001\ncycles=10\n' "$(git rev-parse HEAD)" 'ezcorp:embedding-cache-final-2c542bace8f1' "$(id -g)" > "$receipt/provenance.txt"
cp "$0" "$receipt/frozen-inputs/controller.sh"
for input in scripts/verify-shipping-runtime-resources.sh scripts/verify-shipping-runtime-resources.ts scripts/verify-shipping-runtime.sh scripts/verify-production-image-lifecycle.sh scripts/start-extension-runner-e2e.sh scripts/lib/production-lifecycle-client.ts scripts/lib/shipping-runtime-resource-accounting.ts; do
  mkdir -p "$receipt/frozen-inputs/$(dirname "$input")"
  cp "$input" "$receipt/frozen-inputs/$input"
done
sha256sum "$0" scripts/verify-shipping-runtime-resources.sh scripts/verify-shipping-runtime-resources.ts scripts/verify-shipping-runtime.sh scripts/verify-production-image-lifecycle.sh scripts/start-extension-runner-e2e.sh scripts/lib/production-lifecycle-client.ts scripts/lib/shipping-runtime-resource-accounting.ts /tmp/ez-audit-ci-conmon/podman /tmp/ez-audit-ci-conmon/conmon-2.2.1 > "$receipt/input.sha256"
set +e
EZ_PRODUCTION_IMAGE=ezcorp:embedding-cache-final-2c542bace8f1 EZ_PRODUCTION_APP_UID=1001 EZ_PRODUCTION_APP_GID="$(id -g)" EZ_PRODUCTION_RUNNER_APP_UID=1001 EZ_PRODUCTION_RECEIPT_DIR="$receipt/runtime" EZ_RUNTIME_RESOURCE_CYCLES=10 bash scripts/verify-shipping-runtime-resources.sh > "$receipt/outer.log" 2>&1
rc=$?
set -e
sha256sum -c "$receipt/input.sha256" > "$receipt/input.sha256.verify" 2>&1 || rc=1
printf 'main_exit=%s\n' "$rc" > "$receipt/exit.txt"
exit "$rc"
