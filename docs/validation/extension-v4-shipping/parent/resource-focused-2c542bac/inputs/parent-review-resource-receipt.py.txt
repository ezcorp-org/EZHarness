"""Independently check a completed R4 receipt and its full descriptor samples."""
from pathlib import Path
import argparse
import hashlib
import json
import re

parser = argparse.ArgumentParser()
parser.add_argument("runtime", type=Path)
parser.add_argument("output", type=Path)
parser.add_argument("--minimum-ms", type=int, default=0)
args = parser.parse_args()
root = args.runtime
raw = root / "r4-resource-samples.json"
full = json.loads(raw.read_text())
records = []
for line in (root / "verification.log").read_text().splitlines():
    if line.startswith("{"):
        value = json.loads(line)
        if value.get("check") == "R4":
            records.append(value)
assert len(records) == 1, "Expected exactly one terminal R4 summary"
terminal = records[0]
assert "failure" not in full, "Resource receipt contains a failure"
cycles = terminal["completedCycles"]
assert cycles >= 10 and cycles <= terminal["maximumCycles"]
assert terminal["actualDurationMs"] >= args.minimum_ms
assert terminal["durationMs"] == terminal["actualDurationMs"]
assert terminal["reconnectsPerCycle"] == 10
assert terminal["totalReconnects"] == cycles * 10
if args.minimum_ms:
    assert terminal["mode"] == "duration"
    assert terminal["requestedMinimumDurationMs"] == args.minimum_ms
else:
    assert terminal["mode"] == "cycles"
    assert cycles == terminal["maximumCycles"]
assert full["cycles"] == full["maximumCycles"]
assert terminal["cycles"] == cycles
assert full["actualDurationMs"] == full["durationMs"]
assert args.minimum_ms <= full["actualDurationMs"] <= terminal["actualDurationMs"]
for key in full:
    if key not in {"baseline", "relationCacheWarmup", "samples", "cycles", "actualDurationMs", "durationMs"}:
        assert full[key] == terminal[key], key
samples = full["samples"]
assert len(samples) == len(terminal["samples"]) == cycles + 1
assert [s["cycle"] for s in samples] == list(range(cycles + 1))
assert full["baseline"] == samples[0]
assert full["relationCacheWarmup"] == samples[6]
assert terminal["relationCacheWarmupCycles"] == 6
assert samples[0]["appEstablishedTcpConnections"] == 0
baseline_fds = samples[0]["runnerFds"]
first = samples[1]
first_classes = first["appFdClasses"]
first_other_paths = first_classes["path"] - len(first["pgliteRelationDescriptors"])
relation_count = 0
for sample, summary in zip(samples, terminal["samples"]):
    for key, value in summary.items():
        observed = len(sample.get("pgliteRelationDescriptors", [])) if key == "pgliteRelationDescriptorCount" else sample[key]
        assert observed == value, (sample["cycle"], key)
    assert sample["runnerContainers"] == 0
    assert sample["runnerFds"] == baseline_fds
    if not sample["cycle"]:
        continue
    cleanup = sample["sseCleanup"]
    assert cleanup["remainingConnections"] <= cleanup["baselineConnections"]
    assert cleanup["remainingConnections"] == 0
    assert sample["appEstablishedTcpConnections"] <= cleanup["baselineConnections"]
    classes = sample["appFdClasses"]
    assert sum(classes.values()) == sample["appFds"]
    for key in ["socket", "pipe", "anon", "other"]:
        assert classes[key] <= first_classes[key]
    relations = sample["pgliteRelationDescriptors"]
    assert classes["path"] - len(relations) <= first_other_paths
    identities = set()
    for descriptor in relations:
        assert re.fullmatch(r"/app/data/ezcorp/base/[0-9]+/[0-9]+(?:_(?:fsm|vm|init))?(?:\.[0-9]+)?", descriptor["target"])
        assert not descriptor["backingMissing"]
        assert descriptor["backingDevice"] and descriptor["backingInode"]
        assert descriptor["device"] == descriptor["backingDevice"]
        assert descriptor["inode"] == descriptor["backingInode"]
        identity = (descriptor["device"], descriptor["inode"])
        assert identity not in identities
        identities.add(identity)
        relation_count += 1
warm = samples[6]["appContainerMemoryBytes"]
maximum = max(s["appContainerMemoryBytes"] for s in samples[6:])
assert maximum - warm <= 64 * 1024 ** 2
command = (root / "command.log").read_text()
for field in ["command_exit", "app_log_exit", "owned_cleanup_exit", "verifier_cleanup_exit"]:
    assert re.findall(r"^" + field + r"=([0-9]+)$", command, re.M) == ["0"], field
report = {
    "raw_samples_sha256": hashlib.sha256(raw.read_bytes()).hexdigest(),
    "mode": terminal["mode"], "cycles": cycles,
    "reconnects": terminal["totalReconnects"],
    "actual_duration_ms": terminal["actualDurationMs"],
    "last_sample_receipt_duration_ms": full["actualDurationMs"],
    "required_minimum_ms": args.minimum_ms,
    "samples_verified": len(samples),
    "relation_descriptor_rows_verified": relation_count,
    "runner_fds_each": baseline_fds,
    "owned_workers_each": 0, "remaining_sse_connections_each_cycle": 0,
    "warm_memory_bytes": warm, "maximum_post_warm_memory_bytes": maximum,
    "post_warm_growth_bytes": maximum - warm,
    "final_memory_bytes": samples[-1]["appContainerMemoryBytes"],
    "memory_bound_bytes": 64 * 1024 ** 2,
    "all_four_launcher_exits": 0,
}
args.output.write_text(json.dumps(report, indent=2) + "\n")
print(json.dumps(report, indent=2))
