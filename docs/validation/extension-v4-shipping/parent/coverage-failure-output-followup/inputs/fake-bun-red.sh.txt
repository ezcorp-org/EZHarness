#!/usr/bin/env bash
set -eu
if [[ "${1:-}" == "--version" ]]; then printf '1.3.14\n'; exit 0; fi
args="$*"
if [[ "$args" == *"--coverage"* ]]; then
  if [[ ! -e "$FAKE_BUN_STATE/first" ]]; then
    : > "$FAKE_BUN_STATE/first"
    printf 'FIRST_PASS_FAILURE_SENTINEL: synthetic instrumented failure\n' >&2
    exit 1
  fi
  for arg in "$@"; do
    case "$arg" in --coverage-dir=*) covdir=${arg#--coverage-dir=};; esac
  done
  mkdir -p "$covdir"
  printf 'TN:\nSF:/repo/src/x.ts\nDA:1,1\nLF:1\nLH:1\nend_of_record\n' > "$covdir/lcov.info"
  printf 'COVERAGE_RECOVERY_SENTINEL\n'
  exit 0
fi
printf 'RETRY_SUCCESS_SENTINEL\n'
exit 0
