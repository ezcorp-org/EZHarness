#!/usr/bin/env bash
set -uo pipefail
project=extension-v4-web-audit-docker; compose=/tmp/extension-v4-web-compose.yml; root=/home/dev/work/EZCorp/extension-v4-audit-web
log="$root/docs/validation/extension-v4-independent/web/artifacts/file-organizer-compose.log"; testlog="$root/docs/validation/extension-v4-independent/web/artifacts/file-organizer-real-production.log"; runnerlog="$root/docs/validation/extension-v4-independent/web/artifacts/file-organizer-runner.log"
umask 077; export RUN_ROOT=$(mktemp -d /tmp/ez-file-organizer-runner-XXXXXXXX); mkdir -m 700 "$RUN_ROOT/socket" "$RUN_ROOT/app-data" "$RUN_ROOT/extension-state"; export APP_UID=$(id -u) APP_GID=$(id -g)
export EZ_EXTENSION_RUNNER_SOCKET="$RUN_ROOT/socket/runner.sock" EZ_EXTENSION_RUNNER_TOKEN_FILE="$RUN_ROOT/token" EZ_EXTENSION_RUNNER_STORE="$RUN_ROOT/store" EZ_EXTENSION_APP_UID="$APP_UID"; runner_pid=""
cleanup() { docker compose -p "$project" -f "$compose" logs --no-color > "$log" 2>&1 || true; docker compose -p "$project" -f "$compose" down --volumes --remove-orphans >> "$log" 2>&1 || true; if [ -n "$runner_pid" ]; then kill -TERM "$runner_pid" 2>/dev/null || true; wait "$runner_pid" 2>/dev/null || true; fi; rm -rf "$RUN_ROOT"; }; trap cleanup EXIT
cd "$root"; bash scripts/start-extension-runner-e2e.sh > "$runnerlog" 2>&1 & runner_pid=$!; for i in $(seq 1 120); do [ -S "$EZ_EXTENSION_RUNNER_SOCKET" ] && [ -s "$EZ_EXTENSION_RUNNER_TOKEN_FILE" ] && break; kill -0 "$runner_pid" 2>/dev/null || { cat "$runnerlog"; exit 1; }; sleep 1; done
docker compose -p "$project" -f "$compose" up -d; for i in $(seq 1 90); do curl -fsS http://localhost:3000/api/health >/dev/null 2>&1 && break; sleep 1; done
setup_code=$(curl -sS -o /tmp/extension-v4-web-setup.json -w "%{http_code}" -H "content-type: application/json" --data '<sanitized-setup-payload>' http://localhost:3000/api/auth/setup); echo "setup HTTP $setup_code"; [ "$setup_code" = 201 ] || { cat /tmp/extension-v4-web-setup.json; exit 1; }
cd "$root/web"; DOCKER_TEST=1 EZCORP_APP_CONTAINER=extension-v4-web-audit-app PLAYWRIGHT_BLOB_OUTPUT_FILE="$root/docs/validation/extension-v4-independent/web/artifacts/file-organizer-real-production.zip" ./node_modules/.bin/playwright test e2e/file-organizer-real.spec.ts --project=chromium > "$testlog" 2>&1; code=$?; tail -180 "$testlog"; exit $code
