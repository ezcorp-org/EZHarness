#!/usr/bin/env bash
set -euo pipefail
repo=${1:-$PWD}
image=8f722e76d30f7a4866eb61a2546af64da73f170a5cc9c23866f53ced660e40be
probe=$(mktemp /tmp/ez-seccomp-probe.XXXXXX)
source_file=$(mktemp /tmp/ez-seccomp-test.XXXXXX.ts)
cleanup(){ python3 - "$probe" "$source_file" <<'PY'
from pathlib import Path
import sys
for name in sys.argv[1:]: Path(name).unlink(missing_ok=True)
PY
}
trap cleanup EXIT INT TERM
cp "$repo/src/__tests__/mcp-seccomp-enforce-integration.test.ts" "$source_file"
python3 - "$source_file" <<'PY'
from pathlib import Path
import sys
p=Path(sys.argv[1]); s=p.read_text(); needle='      expect(stderr).toMatch(/probe-io_uring_setup: r=-1 errno=38\\b/);'; p.write_text(s.replace(needle,needle+'\n      console.log(`PROBE_EFFECT=${JSON.stringify(stderr.trim())}`);'))
PY
gcc -O2 -x c -o "$probe" - <<'C'
#include <errno.h>
#include <stdio.h>
#include <sys/syscall.h>
#include <unistd.h>
int main(void){long r=syscall(SYS_getpid);fprintf(stderr,"probe-getpid: r=%ld errno=%d\n",r,errno);errno=0;long denied=syscall(SYS_io_uring_setup,0,NULL);fprintf(stderr,"probe-io_uring_setup: r=%ld errno=%d\n",denied,errno);return 0;}
C
podman run --rm --pull=never --log-driver=k8s-file --cap-add=NET_ADMIN --workdir=/app \
  --env=EZCORP_SECCOMP_PROBE_BIN=/usr/local/bin/seccomp-probe \
  --volume=/nix:/nix:ro \
  --volume="$source_file:/app/src/__tests__/mcp-seccomp-enforce-integration.test.ts:ro" \
  --volume="$repo/src/__tests__/helpers/permission-engine-stub.ts:/app/src/__tests__/helpers/permission-engine-stub.ts:ro" \
  --volume="$probe:/usr/local/bin/seccomp-probe:ro" \
  --entrypoint=/usr/local/bin/bun "$image" test src/__tests__/mcp-seccomp-enforce-integration.test.ts --timeout 30000
