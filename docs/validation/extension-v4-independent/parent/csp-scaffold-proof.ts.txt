// Run from the repository root with Bun 1.3.14 after copying this file to /tmp/ez-parent-csp-proof.ts.
// This is a browser probe of the production scaffold and response headers. It does not
// replace the authenticated route or installed-release browser tests.
import { strict as assert } from "node:assert";
import { join } from "node:path";
const root = process.cwd();
const { chromium } = await import(join(root, "web/node_modules/playwright/index.mjs"));
const { buildScaffold, TAILWIND_BROWSER_SHA256 } = await import(join(root, "docs/extensions/examples/claude-design/lib/generator.ts"));
const { extensionDocumentHeaders } = await import(join(root, "web/src/lib/server/extension-document.ts"));
const output = join(root, "docs/validation/extension-v4-independent/parent");
const html = buildScaffold({
  meta: { schemaVersion: 2, draftId: "audit", prompt: "Standalone canvas proof", kind: "page", createdAt: "2026-09-05T00:00:00Z" },
  designSystem: {
    schemaVersion: 1, source: "greenfield", components: [],
    colors: { primary: "#ff0066", secondary: "#4f46e5", neutral: ["#172032", "#ffffff"] },
    typography: { display: "sans-serif", body: "sans-serif", scale: [16, 24, 40] },
    spacing: { unit: 8, scale: [8, 16, 24, 32] },
  },
  bodyMarkup: `<main id="proof" class="flex items-center min-h-screen" style="padding:calc(var(--space-unit) * 4);background:var(--color-bg);color:var(--color-fg);font-family:var(--font-body)">
    <section style="max-width:640px;margin:auto">
      <p style="color:var(--color-secondary)">EXTENSION CANVAS CHECK</p>
      <h1 style="font-size:var(--font-size-3);line-height:1.1">A self-contained design</h1>
      <p style="font-size:var(--font-size-1);line-height:1.6">The browser renders utility classes and design tokens under the production content security policy.</p>
      <div id="color" class="bg-[var(--color-primary)]" style="height:calc(var(--space-unit) * 8);border-radius:var(--radius-base)"></div>
      <div id="dynamic" style="margin-top:var(--space-unit)">Dynamic utility check</div>
    </section>
  </main>`,
});
const responses: string[] = [];
const server = Bun.serve({ hostname: "127.0.0.1", port: 0, fetch(request) {
  responses.push(new URL(request.url).pathname);
  return new Response(html, { headers: extensionDocumentHeaders() });
}});
const browser = await chromium.launch({ headless: true });
try {
  const page = await browser.newPage({ viewport: { width: 1280, height: 800 } });
  const requests: string[] = [];
  const errors: string[] = [];
  page.on("request", request => requests.push(request.url()));
  page.on("pageerror", error => errors.push(error.message));
  const url = `http://127.0.0.1:${server.port}/scaffold`;
  const response = await page.goto(url);
  assert.equal(response?.status(), 200);
  assert.equal(response?.headers()["content-security-policy"], extensionDocumentHeaders()["Content-Security-Policy"]);
  await page.waitForFunction(() => getComputedStyle(document.querySelector("#proof")!).display === "flex");
  await page.evaluate(() => { document.querySelector("#dynamic")!.className = "grid"; });
  await page.waitForFunction(() => getComputedStyle(document.querySelector("#dynamic")!).display === "grid");
  const actual = await page.evaluate(() => {
    let storageDenied = false;
    try { localStorage.setItem("audit", "test"); } catch (error) { storageDenied = error instanceof DOMException && error.name === "SecurityError"; }
    const proof = getComputedStyle(document.querySelector("#proof")!);
    return { display: proof.display, alignItems: proof.alignItems,
      backgroundColor: getComputedStyle(document.querySelector("#color")!).backgroundColor,
      dynamicDisplay: getComputedStyle(document.querySelector("#dynamic")!).display, storageDenied };
  });
  assert.deepEqual(actual, { display: "flex", alignItems: "center", backgroundColor: "rgb(255, 0, 102)", dynamicDisplay: "grid", storageDenied: true });
  await page.screenshot({ path: join(output, "scaffold-csp-desktop.png"), fullPage: true });
  await page.setViewportSize({ width: 390, height: 844 });
  await page.screenshot({ path: join(output, "scaffold-csp-mobile.png"), fullPage: true });
  assert.deepEqual(requests, [url], "No asset, script, font, or provider network requests are allowed");
  assert.deepEqual(responses, ["/scaffold"]);
  assert.deepEqual(errors, []);
  console.log(JSON.stringify({ status: "passed", actual, requestCount: requests.length,
    remoteAssetRequests: 0, pageErrors: errors, tailwindSha256: TAILWIND_BROWSER_SHA256,
    csp: extensionDocumentHeaders()["Content-Security-Policy"], browser: browser.version(), bun: Bun.version }));
} finally {
  await browser.close();
  server.stop(true);
}
