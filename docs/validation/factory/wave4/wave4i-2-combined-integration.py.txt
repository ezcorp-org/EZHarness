def readings():
 mi = dict(l.split(':', 1) for l in Path('/proc/meminfo').read_text().splitlines()); kib = lambda k: int(mi[k].split()[0])
 return {'memAvailableGiB': round(kib('MemAvailable') / 2**20, 1), 'swapFreeGiB': round(kib('SwapFree') / 2**20, 1), 'diskAvailGB': shutil.disk_usage('/').free // 10**9}
def gate(need_gib, disk_gb, wait_s=1200):
 deadline = time.time() + wait_s
 while True:
  r = readings()
  if r['memAvailableGiB'] >= need_gib and r['swapFreeGiB'] >= 2 and r['diskAvailGB'] >= disk_gb: return r, True
  if time.time() >= deadline: return r, False
  time.sleep(30)
