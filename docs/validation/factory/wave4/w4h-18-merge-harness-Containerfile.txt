# W4H-18 runner-shaped image: a byte copy (comment aside) of w4h-11/image/Containerfile.
# non-root `runner` uid 1001, NOPASSWD sudo, subordinate ids 2..1000 and 1002..65536 under the nesting exception), plus three
# Ubuntu packages the production proofs need: podman-docker (the `docker` CLI is Podman: the Podman-only rule; the hosted job
# uses rootful Docker for the app), docker-compose-v2 (the Compose client the proofs drive) and jq (archived-image builder).
FROM docker.io/library/ubuntu:24.04
ENV DEBIAN_FRONTEND=noninteractive
RUN apt-get update -qq && apt-get install -y -qq --no-install-recommends systemd systemd-sysv dbus dbus-user-session sudo \
      podman uidmap slirp4netns fuse-overlayfs passt python3 util-linux ca-certificates curl git unzip procps \
      podman-docker docker-compose-v2 jq xz-utils >/dev/null \
    && rm -rf /var/lib/apt/lists/* && touch /etc/containers/nodocker
# Node: Playwright (file-organizer proof) needs Node 20 or newer; Ubuntu ships 18.19. The hosted image ships its own Node; this is the
# repository's pin (.node-version 24.14.1, the release setup-node installs in the other jobs), checked against nodejs.org SHASUMS256.
RUN curl -fsSLo /tmp/node.tar.xz https://nodejs.org/dist/v24.14.1/node-v24.14.1-linux-x64.tar.xz \
    && echo "84d38715d449447117d05c3e71acd78daa49d5b1bfa8aacf610303920c3322be  /tmp/node.tar.xz" | sha256sum -c - \
    && tar -xJf /tmp/node.tar.xz -C /usr/local --strip-components=1 --exclude CHANGELOG.md --exclude LICENSE --exclude README.md \
    && rm /tmp/node.tar.xz && node --version
RUN useradd --uid 1001 --create-home --home-dir /home/runner --shell /bin/bash runner && chmod 0755 /home/runner \
    && echo 'runner ALL=(ALL) NOPASSWD:ALL' > /etc/sudoers.d/runner && chmod 0440 /etc/sudoers.d/runner \
    && printf 'runner:2:999\nrunner:1002:64535\n' > /etc/subuid && cp /etc/subuid /etc/subgid \
    && mkdir -p /var/lib/systemd/linger && touch /var/lib/systemd/linger/runner \
    && systemctl mask getty@.service console-getty.service systemd-logind.service.d 2>/dev/null; true
STOPSIGNAL SIGRTMIN+3
CMD ["/sbin/init"]
