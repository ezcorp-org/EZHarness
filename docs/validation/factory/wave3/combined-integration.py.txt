#!/usr/bin/env python3
"""Reusable combined-integration + coverage-gate runner (adapted from the
root-protected-effects-parent-* runners). Usage:
  combined-integration.py --repo PATH --prefix NAME [--skip a,b] [--patch-bases sha,sha]
Writes <evidence>/<prefix>-<check>.log, <prefix>-results.json, <prefix>-coverage-results.json."""
from pathlib import Path
import argparse, os, subprocess, json, fcntl, shutil, sys, datetime, glob
ap = argparse.ArgumentParser()
ap.add_argument('--repo', required=True); ap.add_argument('--prefix', required=True)
ap.add_argument('--skip', default=''); ap.add_argument('--patch-bases', default='644987ada,2588c9f19edcae24273f4a2049eb3ac37bd6f920')
ap.add_argument('--allow-dirty', action='store_true'); ap.add_argument('--podman', action='store_true', help='also run the real Podman suites (serially)'); ap.add_argument('--focused-extra', default=''); ap.add_argument('--reuse-from', default=''); ap.add_argument('--auto-extra-base', default='', help='also run every added/changed *.test.ts (bun-runnable, outside web/ and tests/postgres/) since this ref')
a = ap.parse_args()
root = Path('/tmp/factory-platform-evidence'); repo = Path(a.repo); P = a.prefix
skip = set(filter(None, a.skip.split(',')))
MEASURED = {'sdk': ['packages/@ezcorp/factory-sdk'], 'product': ['src','packages','scripts'], 'postgres': ['src','tests','packages'], 'node': ['packages/@ezcorp/factory-orchestrator','src/factory'], 'web': ['web/src'], 'pool': ['src/factory/pool','src/factory','tests/postgres'], 'compute': ['src/factory','tests/postgres'], 'provisioning': ['src/factory/provisioning','src/factory','tests/postgres'], 'python': ['src/factory/runner/python']}
PRODUCER_CHECK = {'sdk': 'sdk-tests', 'product': 'focused', 'postgres': 'postgres', 'node': 'node-coverage', 'web': 'web-coverage', 'pool': 'pool-coverage', 'compute': 'compute-coverage', 'provisioning': 'provisioning-coverage', 'python': 'python-coverage'}
env = os.environ.copy(); env['PATH'] = '/tmp/factory-tools/bun-1.3.14/bun-linux-x64:' + env['PATH']
fields = dict(l.split('=', 1) for l in (root / 'postgres.env').read_text().splitlines())
endpoint = subprocess.check_output(['podman', 'port', 'factory-platform-proof-postgres', '5432'], text=True).strip()
env.pop('DATABASE_URL', None); env.pop('GATE_CHANGE_APPROVED', None)
env['FACTORY_TEST_POSTGRES_URL'] = f"postgres://{fields['POSTGRES_USER']}:{fields['POSTGRES_PASSWORD']}@{endpoint}/{fields['POSTGRES_DB']}"
env['EZCORP_FACTORY_STORAGE_SECRETS_DIR'] = '/run/user/1001/ezcorp-factory-storage.8yWJyCIQ'
env['FACTORY_TEMPORAL_TEST_SERVER'] = '/tmp/factory-tools/temporal-test-server/temporal-test-server_1.38.0_linux_amd64/temporal-test-server'
workflow = (repo / '.github/workflows/db-postgres.yml').read_text()
ci_files = [l.strip() for l in workflow.splitlines() if l.strip().startswith('./tests/postgres/factory-')]
assert len(ci_files) >= 20, ci_files
focused = ['./src/__tests__/factory-run-lifecycle.test.ts','./src/factory/release-authority.integration.test.ts','./src/factory/executions.integration.test.ts','./src/factory/run-inputs.integration.test.ts','./src/db/migrations/add-factory-child-runs.test.ts','./src/db/migrations/bind-factory-child-definition-source.test.ts','./src/db/migrations/bind-factory-child-start-clock.test.ts','./scripts/factory-task-execution-admission-registration.test.ts','./src/db/migrations/add-factory-command-approvals.test.ts','./src/factory/release-application.test.ts','./src/factory/application.test.ts','./src/factory/private-commands.test.ts','./src/__tests__/factory-private-service.test.ts','./src/factory/validator-materials.test.ts','./src/factory/assurance.test.ts','./src/factory/releases.integration.test.ts','./src/factory/attempt-queue.integration.test.ts','./src/delivery-queue/durable-delivery-queue.test.ts','./scripts/factory-validator-materials-registration.test.ts','./src/factory/package-preparation.integration.test.ts','./src/db/migrations/add-factory-package-preparations.test.ts','./src/db/migrations/add-factory-task-outcomes.test.ts','./src/factory/artifacts.integration.test.ts','./src/factory/protected-command-provenance.test.ts','./scripts/factory-protected-command-effects-registration.test.ts','./src/db/factory-schema.test.ts','./src/__tests__/gate-scripts.test.ts']
ALWAYS_EXTRA = ['./src/extensions/__tests__/project-pull-request-transport.test.ts','./src/extensions/__tests__/github-projects-handler.test.ts','./src/extensions/__tests__/github-projects-handler.integration.test.ts','./src/extensions/__tests__/github-projects-handler-default-db.test.ts','./src/extensions/__tests__/tool-executor.github-projects-rpc.test.ts','./src/factory/release-adapters.test.ts','./src/__tests__/workflow-release-assets.test.ts','./src/__tests__/workflow-release-consent.test.ts','./src/__tests__/workflow-service-release-authority.test.ts','./src/__tests__/workflow-run-persistence.test.ts','./src/__tests__/workflow-nesting.test.ts']
auto = []
if a.auto_extra_base:
 changed = subprocess.check_output(['git','diff','--name-only','--diff-filter=AM',a.auto_extra_base,'HEAD'], cwd=repo, text=True).split()
 auto = ['./'+f for f in changed if f.endswith('.test.ts') and not f.startswith(('web/','tests/postgres/')) and '.podman.' not in f and 'temporal' not in f and (repo/f).exists()]
focused = list(dict.fromkeys(f for f in focused + ALWAYS_EXTRA + auto + list(filter(None, a.focused_extra.split(','))) if (repo / f).exists()))
print('focused files:', len(focused), 'auto-added:', len(auto), flush=True)
web_tests = [f for f in ['src/lib/factory/FactoryReleaseInbox.component.test.ts','src/lib/factory/client.unit.test.ts','src/routes/(app)/factories/factories-page.server.test.ts','src/routes/api/factories/factories.server.test.ts'] if (repo / 'web' / f).exists()]
cov = lambda n: str(root / f'{P}-{n}-coverage')
checks = [
 ('sdk-build', ['bun','run','--cwd','packages/@ezcorp/factory-sdk','build']),
 ('transport-build', ['bun','run','--cwd','packages/@ezcorp/factory-transport','build']),
 ('sdk-tests', ['bun','test','--timeout','30000','--coverage','--coverage-reporter=lcov',f'--coverage-dir={cov("sdk")}','./packages/@ezcorp/factory-sdk/src']),
 ('focused', ['bun','test','--timeout','30000','--coverage','--coverage-reporter=lcov',f'--coverage-dir={cov("product")}',*focused]),
 ('web-coverage', ['bash','scripts/web-vitest-coverage.sh','--output',cov('web')]),
 ('postgres', ['bun','test','--timeout','60000','--coverage','--coverage-reporter=lcov',f'--coverage-dir={cov("postgres")}',*ci_files]),
 ('types', ['bun','run','typecheck']),
 ('lint', ['bun','run','lint']),
 ('gate-integrity', ['bun','scripts/gate-integrity.ts']),
 ('boundaries', ['bun','scripts/check-factory-boundaries.ts']),
 ('orchestrator-build', ['bun','run','--cwd','packages/@ezcorp/factory-orchestrator','build']),
 ('node-coverage', ['bash','scripts/factory-orchestrator-coverage.sh']),
 ('pool-coverage', ['bash','scripts/factory-pool-coverage.sh']),
 ('compute-coverage', ['bash','scripts/factory-compute-admissions-coverage.sh']),
 ('provisioning-coverage', ['bash','scripts/factory-provisioning-coverage.sh']),
 ('python-coverage', ['nix-shell','-p','uv','--run','bash scripts/python-quality.sh all']),
]
if a.podman:
 for f in ['./packages/@ezcorp/extension-runner/tests/podman.integration.test.ts','./src/factory/runner/attempt-runtime.integration.test.ts','./src/factory/runner/supervisor.podman.integration.test.ts','./src/factory/package-preparation.podman.integration.test.ts']:
  if (repo / f).exists(): checks.append(('podman-' + Path(f).stem.split('.')[0], ['bun','test','--timeout','300000',f]))
with open('/tmp/ezcorp-validation-heavy.lock', 'w') as lock:
 dirty = subprocess.check_output(['git','status','--porcelain'], cwd=repo, text=True)
 if dirty and not a.allow_dirty: sys.exit(f'dirty tree:\n{dirty}')
 fcntl.flock(lock, fcntl.LOCK_EX)
 sha = subprocess.check_output(['git','rev-parse','HEAD'], cwd=repo, text=True).strip()
 reuse = [r for r in a.reuse_from.split(',') if r]
 def reusable(label):
  for prior_prefix in reuse:
   alt = root / f'{prior_prefix}-{label}-coverage' / 'lcov.info'
   if not alt.exists(): continue
   prior_head = json.loads((root / f'{prior_prefix}-results.json').read_text())[0]['head']
   if subprocess.run(['git','diff','--quiet',prior_head,sha,'--',*MEASURED[label]], cwd=repo).returncode == 0: return (alt, prior_head)
  return None
 for label, check in PRODUCER_CHECK.items():
  if check in skip and 'coverage' not in skip and not reusable(label):
   print(f'reuse of {label} coverage is not valid at {sha[:9]}; running {check} instead', flush=True); skip.discard(check)
   if check == 'node-coverage': skip.discard('orchestrator-build')
 results = []; failed = False
 for name, cmd in checks:
  if name in skip: continue
  e = env.copy()
  if name == 'postgres': e['DATABASE_URL'] = e['FACTORY_TEST_POSTGRES_URL']
  if name == 'node-coverage': e['COV_OUT'] = cov('node')
  if name in ('pool-coverage','compute-coverage','provisioning-coverage'):
   e['COV_OUT'] = cov(name.split('-')[0]); e['DATABASE_URL'] = e['FACTORY_TEST_POSTGRES_URL']
  if name == 'python-coverage': e['PYTHON_COVERAGE_OUT'] = cov('python')
  start = datetime.datetime.now().isoformat(timespec='seconds')
  with (root / f'{P}-{name}.log').open('w') as out:
   try: r = subprocess.run(cmd, cwd=repo, env=e, stdout=out, stderr=subprocess.STDOUT, timeout=1800)
   except (OSError, subprocess.TimeoutExpired) as err:
    out.write(type(err).__name__ + '\n'); r = subprocess.CompletedProcess(cmd, 124 if isinstance(err, subprocess.TimeoutExpired) else 127)
  rec = {'head': sha, 'dirty': bool(dirty), 'name': name, 'exitCode': r.returncode, 'start': start, 'end': datetime.datetime.now().isoformat(timespec='seconds'), 'command': cmd}
  results.append(rec); (root / f'{P}-results.json').write_text(json.dumps(results, indent=1) + '\n'); print(json.dumps({k: rec[k] for k in ('name','exitCode','start','end')}), flush=True)
  if r.returncode: failed = True
 # coverage merge + gates (only if every producer passed)
 covres = {'head': sha, 'producers': [], 'checks': []}
 if not failed and 'coverage' not in skip:
  inputs = root / f'{P}-lcov-inputs'; shutil.rmtree(inputs, ignore_errors=True); inputs.mkdir()
  for label in ['sdk','product','postgres','node','pool','compute','provisioning','python']:
   src = Path(cov(label)) / 'lcov.info'
   if src.exists(): covres['producers'].append({'producer': label, 'input': str(src), 'head': sha})
   else:
    r = reusable(label)
    if r: src, prior_head = r; covres['producers'].append({'producer': label, 'input': str(src), 'head': prior_head, 'reused': True, 'sourceIdenticalTo': sha, 'paths': MEASURED[label]})
   if src.exists(): shutil.copyfile(src, inputs / f'{label}.lcov')
  web = Path(cov('web')) / 'lcov.info'
  if not web.exists():
   r = reusable('web')
   if r: web, prior_head = r; covres['producers'].append({'producer': 'web-vitest', 'input': str(web), 'head': prior_head, 'reused': True, 'sourceIdenticalTo': sha, 'paths': MEASURED['web']})
  if web.exists():
   (inputs / 'web.lcov').write_text('\n'.join(('SF:web/' + l[3:]) if l.startswith('SF:') and not l.startswith('SF:web/') and not l.startswith('SF:/') else l for l in web.read_text().splitlines()) + '\n')
   covres['producers'].append({'producer': 'web-vitest', 'input': str(web), 'head': sha, 'rerooted': True})
  out = repo / 'coverage' / 'lcov.info'; out.parent.mkdir(exist_ok=True)
  with (root / f'{P}-coverage-merge.log').open('w') as log:
   m = subprocess.run(['bun','scripts/merge-lcov.ts',str(inputs / '*.lcov'),str(out)], cwd=repo, env=env, stdout=log, stderr=subprocess.STDOUT)
  covres['checks'].append({'name':'merge','exitCode':m.returncode})
  if m.returncode: failed = True
  if m.returncode == 0:
   for base in filter(None, a.patch_bases.split(',')):
    for label, script in [('new-file','scripts/check-new-file-coverage.ts'),('patch','scripts/check-patch-coverage.ts')]:
     e = env.copy(); e['BASE_REF'] = base
     with (root / f'{P}-coverage-{label}-{base[:9]}.log').open('w') as log:
      r = subprocess.run(['bun',script], cwd=repo, env=e, stdout=log, stderr=subprocess.STDOUT)
     covres['checks'].append({'name': label, 'base': base, 'exitCode': r.returncode}); print(json.dumps(covres['checks'][-1]), flush=True)
     if r.returncode: failed = True
  shutil.copyfile(out, root / f'{P}-merged-lcov.info')
 (root / f'{P}-coverage-results.json').write_text(json.dumps(covres, indent=1) + '\n')
 print('DONE failed=%s' % failed, flush=True)
